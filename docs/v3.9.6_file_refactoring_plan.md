```python
class RobotController:
    """RoArm-M3 제어"""
    
    def __init__(self, robot: Articulation, gripper: Gripper):
        self.robot = robot
        self.gripper = gripper
        
        # Joint 인덱스
        self.arm_joint_indices = [0, 1, 2, 3, 4, 5]
        self.gripper_joint_indices = [6, 7]
        
        # 그리퍼 증분 제어
        self.current_gripper_pos = 0.0  # rad
        self.current_gripper_width = 0.03  # m (BASE_GAP)
    
    def apply_action(self, action: np.ndarray):
        """액션 적용 (7차원)"""
        # 1. Arm: Joint velocities (6차원)
        arm_velocities = action[:6] * 0.5  # scale
        
        # 2. Gripper: 증분 제어 (1차원)
        gripper_scalar = action[6]
        gripper_delta = gripper_scalar * 0.01  # ±0.01 rad/step
        
        # Gripper position 업데이트
        self.current_gripper_pos = np.clip(
            self.current_gripper_pos + gripper_delta,
            0.0,    # 완전 닫힘
            0.025   # 완전 열림 (URDF limit)
        )
        
        # Width 계산 (URDF 베이스 간격 포함)
        BASE_GAP = 0.03
        self.current_gripper_width = BASE_GAP + (self.current_gripper_pos * 2.0)
        
        # Target positions 조합
        target_positions = np.concatenate([
            arm_velocities,
            [self.current_gripper_pos, self.current_gripper_pos]
        ])
        
        # Articulation에 적용
        action_obj = ArticulationAction(joint_positions=target_positions)
        self.robot.apply_action(action_obj)
    
    def reset(self, joint_positions: np.ndarray):
        """관절 리셋"""
        self.current_gripper_pos = 0.0
        self.current_gripper_width = 0.03
        
        # Joint positions 설정
        self.robot.set_joint_positions(joint_positions)
    
    def get_gripper_width(self) -> float:
        """그리퍼 폭 반환"""
        return self.current_gripper_width
```

```python
class RoArmPickPlaceEnv:
    """RoArm-M3 Pick & Place 환경 (v3.9.6 리팩토링)"""
    
    def __init__(self, cfg: RoArmPickPlaceEnvCfg):

        """환경 리셋"""

        self.controller.reset(initial_joints)
        
        # 2. 물체 위치 랜덤화
        cube_pos = self._randomize_cube_position()
        target_pos = self._randomize_target_position()
        
        # 3. World 스텝
        self.world.step(render=True)
        for _ in range(10):
            self.world.step(render=False)
        
        # 4. 로깅 리셋
        self.reward_calc.reset()
        self.step_count = 0
        self.episode_reach_count = 0
        ...
        
        # 5. 관측 반환
        obs = self.obs_builder.build_observation()
        return obs
    
    def step(self, action):
        """환경 스텝"""
        # 1. 액션 적용
        self.controller.apply_action(action)
        
        # 2. 시뮬레이션 스텝
        self.world.step(render=True)
        
        # 3. Attach/Detach 로직
        self._handle_attach_detach()
        
        # 4. 관측 생성
        obs = self.obs_builder.build_observation()
        
        # 5. 보상 계산
        dist_to_cube = self._get_distance_to_cube()
        gripper_width = self.controller.get_gripper_width()
        is_attached = self.gripper.is_attached()
        
        reward = self.reward_calc.calculate_reward(
            obs, is_attached, dist_to_cube, gripper_width
        )
        
        # 6. 종료 조건
        done = self._is_done()
        
        # 7. Info 생성
        info = self._build_info()
        
        self.step_count += 1
        return obs, reward, done, info
    
    def _handle_attach_detach(self):
        """Attach/Detach 로직"""
        dist = self._get_distance_to_cube()
        gripper_width = self.controller.get_gripper_width()
        
        is_grasping = self.gripper.is_grasped(
            self.cube, dist, gripper_width
        )
        
        if is_grasping and not self.gripper.is_attached():
            self.gripper.attach(self.cube)
            print(f"  ✅ Attach! step={self.step_count}")
        
        elif not is_grasping and self.gripper.is_attached():
            self.gripper.detach()
            print(f"  ❌ Detach! step={self.step_count}")
```
